## Implementation Steps

### Step 1: Create Module `json_candles.py`

```python
import json
import yfinance as yf
from datetime import datetime, timedelta
import os

class JSONCandleManager:
    def __init__(self, stocks_file="stocks.json", candles_file="candles_4days.json"):
        self.stocks_file = stocks_file
        self.candles_file = candles_file
        self.stocks = {}
        self.candles = {}
    
    # === BOOTSTRAP (First time only) ===
    def bootstrap_candles(self, stock_list):
        """
        Fetch 4 days of 5-min candles from Yahoo Finance (ONE TIME at 9:15 AM)
        Time: 20-30 seconds for 600 stocks
        """
        print("🔄 Bootstrapping 600 stocks from Yahoo Finance (one-time)...")
        
        self.candles = {}
        end_date = datetime.now()
        start_date = end_date - timedelta(days=4)
        
        for i, stock in enumerate(stock_list):
            try:
                data = yf.download(
                    stock, 
                    start=start_date, 
                    end=end_date, 
                    interval="5m",
                    progress=False
                )
                
                # Group by date
                for date, group in data.groupby(data.index.date):
                    date_str = str(date)
                    if date_str not in self.candles:
                        self.candles[date_str] = {}
                    
                    for idx, row in group.iterrows():
                        time_str = idx.strftime("%H_%M_%H_%M")  # e.g., "9_15_9_20"
                        if time_str not in self.candles[date_str]:
                            self.candles[date_str][time_str] = {}
                        
                        self.candles[date_str][time_str][stock] = {
                            "open": round(float(row['Open']), 2),
                            "high": round(float(row['High']), 2),
                            "low": round(float(row['Low']), 2),
                            "close": round(float(row['Close']), 2),
                            "volume": int(row['Volume']) if 'Volume' in row else 0
                        }
                
                if (i + 1) % 50 == 0:
                    print(f"✅ Fetched {i + 1}/600 stocks...")
            
            except Exception as e:
                print(f"⚠️ Error fetching {stock}: {e}")
        
        self.save_candles()
        print("✅ Bootstrap complete! Data saved to candles_4days.json")
    
    # === LOAD FROM DISK ===
    def load_stocks(self):
        """Load stock list with tokens"""
        try:
            with open(self.stocks_file, 'r') as f:
                self.stocks = json.load(f)
            print(f"✅ Loaded {len(self.stocks)} stocks")
        except FileNotFoundError:
            print(f"❌ {self.stocks_file} not found")
    
    def load_candles(self):
        """Load candle data from disk"""
        try:
            with open(self.candles_file, 'r') as f:
                self.candles = json.load(f)
            print(f"✅ Loaded candle data")
        except FileNotFoundError:
            print(f"⚠️ {self.candles_file} not found, will create on bootstrap")
    
    def save_candles(self):
        """Save candle data to disk (async-friendly)"""
        try:
            with open(self.candles_file, 'w') as f:
                json.dump(self.candles, f)
        except Exception as e:
            print(f"❌ Error saving candles: {e}")
    
    # === REAL-TIME WEBSOCKET UPDATES ===
    def update_candle_from_tick(self, stock, price, volume=0):
        """
        Update current 5-min candle from WebSocket tick
        Called on every tick arrival
        """
        today = datetime.now().strftime("%Y-%m-%d")
        current_5min = self.get_current_5min_slot()
        
        if today not in self.candles:
            self.candles[today] = {}
        
        if current_5min not in self.candles[today]:
            self.candles[today][current_5min] = {}
        
        if stock not in self.candles[today][current_5min]:
            # First tick - initialize
            self.candles[today][current_5min][stock] = {
                "open": price,
                "high": price,
                "low": price,
                "close": price,
                "volume": volume
            }
        else:
            # Update existing
            candle = self.candles[today][current_5min][stock]
            candle["high"] = max(candle["high"], price)
            candle["low"] = min(candle["low"], price)
            candle["close"] = price
            candle["volume"] = volume
        
        self.save_candles()  # Async batch save
    
    # === FALLBACK: Yahoo Finance for missing candles ===
    def fill_missing_candles(self, stock, date):
        """
        If WebSocket data is missing for a stock/date,
        fetch from Yahoo Finance and store
        """
        try:
            data = yf.download(
                stock,
                start=date,
                end=date + timedelta(days=1),
                interval="5m",
                progress=False
            )
            
            date_str = str(date)
            if date_str not in self.candles:
                self.candles[date_str] = {}
            
            for idx, row in data.iterrows():
                time_str = idx.strftime("%H_%M_%H_%M")
                if time_str not in self.candles[date_str]:
                    self.candles[date_str][time_str] = {}
                
                self.candles[date_str][time_str][stock] = {
                    "open": round(float(row['Open']), 2),
                    "high": round(float(row['High']), 2),
                    "low": round(float(row['Low']), 2),
                    "close": round(float(row['Close']), 2),
                    "volume": int(row['Volume']) if 'Volume' in row else 0
                }
            
            self.save_candles()
            print(f"✅ Filled missing candles for {stock} on {date}")
        
        except Exception as e:
            print(f"❌ Error filling candles for {stock}: {e}")
    
    # === UTILITY FUNCTIONS ===
    def get_current_5min_slot(self):
        """Get current 5-min timeframe slot (e.g., '9_15_9_20')"""
        now = datetime.now()
        hour = now.hour
        minute = now.minute
        
        # Round down to nearest 5-min
        slot_start = (minute // 5) * 5
        slot_end = slot_start + 5
        
        return f"{hour}_{slot_start}_{hour}_{slot_end}"
    
    def get_candle(self, stock, date, timeframe):
        """Get specific candle"""
        date_str = str(date)
        try:
            return self.candles[date_str][timeframe].get(stock)
        except:
            return None
    
    def calculate_gap_up(self, stock, gap_pct=2):
        """Check if stock gapped up by X%"""
        today = datetime.now().strftime("%Y-%m-%d")
        yesterday = (datetime.now() - timedelta(days=1)).strftime("%Y-%m-%d")
        
        # Get yesterday's close (last candle of day)
        yesterday_close = None
        if yesterday in self.candles:
            last_candle_key = list(self.candles[yesterday].keys())[-1]
            yesterday_close = self.candles[yesterday][last_candle_key].get(stock, {}).get("close")
        
        # Get today's open (first candle)
        first_candle_key = list(self.candles[today].keys())[0]
        today_open = self.candles[today][first_candle_key].get(stock, {}).get("open")
        
        if yesterday_close and today_open:
            gap = ((today_open - yesterday_close) / yesterday_close) * 100
            return gap >= gap_pct, gap
        
        return False, 0
    
    def calculate_200_ema(self, stock):
        """Calculate 200 EMA from last 200 candles"""
        all_closes = []
        
        # Collect closes from all dates/timeframes
        for date in sorted(self.candles.keys())[-4:]:  # Last 4 days
            for timeframe in sorted(self.candles[date].keys()):
                if stock in self.candles[date][timeframe]:
                    all_closes.append(self.candles[date][timeframe][stock]["close"])
        
        if len(all_closes) < 200:
            return None
        
        # Simple EMA calculation
        ema = all_closes[-200]
        multiplier = 2 / (200 + 1)
        
        for price in all_closes[-199:]:
            ema = price * multiplier + ema * (1 - multiplier)
        
        return round(ema, 2)
    
    def get_all_candles_for_table(self, date):
        """Get all stocks' candles for display in table"""
        date_str = str(date)
        result = []
        
        if date_str not in self.candles:
            return result
        
        # Get latest timeframe
        latest_timeframe = sorted(self.candles[date_str].keys())[-1]
        
        for stock in self.stocks.keys():
            if stock in self.candles[date_str][latest_timeframe]:
                candle = self.candles[date_str][latest_timeframe][stock]
                gap_up, gap_pct = self.calculate_gap_up(stock)
                ema_200 = self.calculate_200_ema(stock)
                
                result.append({
                    "Stock": stock,
                    "Open": candle["open"],
                    "High": candle["high"],
                    "Low": candle["low"],
                    "Close": candle["close"],
                    "Volume": candle["volume"],
                    "Gap %": round(gap_pct, 2) if gap_pct else 0,
                    "Gap Up": "✅" if gap_up else "❌",
                    "EMA 200": ema_200 if ema_200 else "N/A"
                })
        
        return result
```

import streamlit as st
from json_candles import JSONCandleManager
from datetime import datetime

# === INITIALIZATION (Run once) ===
@st.cache_resource
def load_json_manager():
    manager = JSONCandleManager()
    manager.load_stocks()
    manager.load_candles()
    return manager

manager = load_json_manager()

# === BOOTSTRAP BUTTON (First time only) ===
if st.button("🔄 Bootstrap 600 Stocks (Do this once at 9:15 AM)"):
    stock_list = list(manager.stocks.keys())
    manager.bootstrap_candles(stock_list)
    st.success("✅ Bootstrap complete!")

# === REPLACE TRADINGVIEW WITH JSON ===
# In your MomentumScanner or other pages:

def display_stocks_from_json():
    """Display stocks table from JSON (replaces TradingView)"""
    today = datetime.now().strftime("%Y-%m-%d")
    
    df = manager.get_all_candles_for_table(today)
    
    st.dataframe(
        df,
        use_container_width=True,
        hide_index=True
    )

# === CALL IN YOUR SCANNING PAGES ===

display_stocks_from_json()

## Workflow Summary

### 9:15 AM (Market Open)
1. Load `stocks.json` and `candles_4days.json` (instant)
2. **First time only:** Click "Bootstrap" button → Fetch 4 days from Yahoo (20-30 sec)
3. Ready to trade ✅

### 9:20 AM - 3:30 PM (Trading Hours)
1. WebSocket ticks arrive
2. Update JSON candles in real-time (~1-2ms per update)
3. Display table from JSON (NO TradingView calls)

### If WebSocket Data Missing
1. Detect gap in candles
2. Auto-fetch from Yahoo for that timeframe
3. Store in JSON and continue

### Next Day
1. Load JSON (has 4 days)
2. Bootstrap fetch only if needed
3. Continue with WebSocket
